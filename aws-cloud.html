# HighTempAlertFn — AWS Lambda · TECH 300 IoT Demo import boto3, json, os sns_client = boto3.client('sns') TOPIC_ARN = os.environ['SNS_TOPIC_ARN'] def lambda_handler(event, context): device_id = event.get('device_id', 'unknown') temp = event.get('temp', 0) timestamp = event.get('timestamp', 0) # Guard: only alert when temp strictly above threshold if temp <= 30.0: return {'statusCode': 200, 'body': 'Below threshold'} message = ( f"CRITICAL ALERT: High temperature detected on {device_id}!\n" f"Temperature: {temp:.1f}°C (threshold: 30.0°C)\n" f"Timestamp: {timestamp}\n" f"Action Required: Inspect node and activate cooling." ) sns_client.publish( TopicArn = TOPIC_ARN, Subject = f"[IoT Alert] High Temp — {device_id}", Message = message ) return {'statusCode': 200, 'body': 'SNS alert dispatched'}
